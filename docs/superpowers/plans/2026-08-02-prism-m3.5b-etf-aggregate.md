package aggregate

// HarmonicPE 是 D3 口径纯函数:剔除 pe≤0 或 NaN 的成分,权重重归一,
// 返回加权调和平均 PE 与被剔除权重占比。全剔除时 pe=NaN。
func HarmonicPE(weights, pes []float64) (pe, excludedWeight float64)

// PELookup 为一只成分股解析当前 PE:优先库内 valuation_daily 最新值(≤3 个交易日内),
// 缺失走 quote 兜底(yahoo trailingPE,Task 1 of M3.5a 已注明 ⚠ live 校验)。
type PELookup func(ticker string) float64 // 不可得返回 NaN

// Aggregate 对一份持仓快照算当日聚合行。
func Aggregate(h *edgar.HoldingsSnapshot, lookup PELookup) (pe, excludedWeight float64)

- [ ] **Step 1: 失败测试**（纯函数,精确断言）:

```go
// Context Checkpoint: D3 口径:
//   functional[0] 调和平均+重归一          → TestHarmonicPE
//   boundary[0]   负 PE/NaN 剔除计入 excluded → TestHarmonicPEExcludes
//   boundary[1]   全剔除 → NaN + excluded=1   → TestHarmonicPEAllExcluded
func TestHarmonicPE(t *testing.T) {
	// 两只等权 PE 10/30 → 调和 = 1/(0.5/10+0.5/30) = 15
	pe, ex := HarmonicPE([]float64{0.5, 0.5}, []float64{10, 30})
	assert.InDelta(t, 15.0, pe, 1e-9)
	assert.InDelta(t, 0.0, ex, 1e-9)
}
func TestHarmonicPEExcludes(t *testing.T) {
	// 0.5/0.3/0.2 权重,PE 10/-5/NaN → 剔除后重归一只剩第一只 → PE=10, excluded=0.5
	pe, ex := HarmonicPE([]float64{0.5, 0.3, 0.2}, []float64{10, -5, math.NaN()})
	assert.InDelta(t, 10.0, pe, 1e-9)
	assert.InDelta(t, 0.5, ex, 1e-9)
}
```

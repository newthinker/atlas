// —— 验证者自己的响应构造（不复用 dev 的 tabsWithIDs/tabsResponses，免得共享同一个缺陷）——

const vTplID = 777 // 模板 2024年 的 sheetId，刻意与 dev 夹具的 12345 不同

// vTabs 造 spreadsheets.get 响应：按传入顺序给 id 10,11,12…，「2024年」固定 vTplID。
func vTabs(titles ...string) string {
	parts := make([]string, len(titles))
	for i, t := range titles {
		id := 10 + i
		if t == "2024年" {
			id = vTplID
		}
		parts[i] = fmt.Sprintf(`{"properties":{"sheetId":%d,"title":%q,"index":%d}}`, id, t, i)
	}
	return `{"sheets":[` + strings.Join(parts, ",") + `]}`
}

// vHeaders 给若干表配齐表头（录入区不配 ⇒ 服务端回 {} ⇒ 空，正是新建表的样子）。
func vHeaders(m map[string]string, tabs ...string) map[string]string {
	for _, tab := range tabs {
		m["/v4/spreadsheets/sheet-id/values/'"+tab+"'!A3:AI3"] = `{"values":[["月份","发布日期","社融存量","M2余额"]]}`
	}
	return m
}

// vResp：列出 titles、模板 A1 给 tplTitle、为 headerTabs 配表头。
func vResp(tplTitle string, titles []string, headerTabs ...string) map[string]string {
	m := map[string]string{
		pathTabs: vTabs(titles...),
		"/v4/spreadsheets/sheet-id/values/'2024年'!A1": fmt.Sprintf(`{"values":[[%q]]}`, tplTitle),
	}
	if tplTitle == "" {
		m["/v4/spreadsheets/sheet-id/values/'2024年'!A1"] = `{}`
	}
	return vHeaders(m, append(append([]string{}, titles...), headerTabs...)...)
}

// —— batchUpdate 请求体的结构化解析 ——

type vReq struct {
	DuplicateSheet *struct {
		SourceSheetID int64 `json:"sourceSheetId"`
		NewSheetID    int64 `json:"newSheetId"`
	} `json:"duplicateSheet"`
	UpdateSheetProperties *struct {
		Fields     string `json:"fields"`
		Properties struct {
			SheetID int64  `json:"sheetId"`
			Title   string `json:"title"`
			Index   int64  `json:"index"`
		} `json:"properties"`
	} `json:"updateSheetProperties"`
	UpdateCells *struct {
		Fields string `json:"fields"`
		Range  struct {
			SheetID       int64 `json:"sheetId"`
			StartRowIndex int64 `json:"startRowIndex"`
			EndRowIndex   int64 `json:"endRowIndex"`
		} `json:"range"`
		Rows []struct {
			Values []struct {
				UserEnteredValue struct {
					StringValue string `json:"stringValue"`
				} `json:"userEnteredValue"`
			} `json:"values"`
		} `json:"rows"`
	} `json:"updateCells"`
}

// vBatchUpdates 解析出全部 spreadsheets:batchUpdate 的请求列表（按发出顺序）。
func vBatchUpdates(t *testing.T, rec *recorder) [][]vReq {
	t.Helper()
	var out [][]vReq
	// 🔴 判据必须是**精确路径**：建表打 `<ss>:batchUpdate`，写数据打 `<ss>/values:batchUpdate`，
	// 用 HasSuffix(":batchUpdate") 会把数据写请求也算进来（验证者首版就栽在这里）。
	for i, m := range rec.methods {
		if m != "POST "+pathBatchUpdate {
			continue
		}
		var body struct {
			Requests []vReq `json:"requests"`
		}
		require.NoError(t, json.Unmarshal([]byte(rec.bodies[i]), &body), "batchUpdate 请求体应是合法 JSON")
		out = append(out, body.Requests)
	}
	return out
}

// vNewTabIndexes 取每次建表 batchUpdate 里「④ 改 index」那一步的 index 值，按发出顺序。
func vNewTabIndexes(t *testing.T, rec *recorder) []int64 {
	t.Helper()
	var got []int64
	for _, reqs := range vBatchUpdates(t, rec) {
		for _, r := range reqs {
			if r.UpdateSheetProperties != nil && r.UpdateSheetProperties.Fields == "index" {
				got = append(got, r.UpdateSheetProperties.Properties.Index)
			}
		}
	}
	return got
}

// W1 —— 四步：结构化逐字段核，而不是「拼接串里出现过这几个字」。
func TestW_CreateYearTabFourStepsStructurally(t *testing.T) {
	c, rec := newTestClient(t, vResp("2024 年 · 金融数据追踪", []string{"说明", "2024年", "2026年"}), 0)

	require.NoError(t, c.CreateYearTab(context.Background(), "2024年", "2021年", 2021, 3))

	batches := vBatchUpdates(t, rec)
	require.Len(t, batches, 1, "四步必须在**一次** batchUpdate 里（API 原子）")
	require.Len(t, writeBodies(rec), 1, "整个过程只发一个写请求")
	reqs := batches[0]
	require.Len(t, reqs, 4, "恰四步，不多不少")

	// 新表 id = 现有最大 id + 1。现有 {说明:10, 2024年:777, 2026年:12} ⇒ max=777 ⇒ new=778
	const wantNewID int64 = vTplID + 1

	// ① 复制模板
	require.NotNil(t, reqs[0].DuplicateSheet, "第 1 步必须是 duplicateSheet")
	require.Equal(t, int64(vTplID), reqs[0].DuplicateSheet.SourceSheetID, "复制的是模板表的 id")
	require.Equal(t, wantNewID, reqs[0].DuplicateSheet.NewSheetID, "新表 id = 现有最大 + 1")

	// ② 改名
	require.NotNil(t, reqs[1].UpdateSheetProperties, "第 2 步必须是改属性")
	require.Equal(t, "title", reqs[1].UpdateSheetProperties.Fields, "fields 必须限定为 title")
	require.Equal(t, "2021年", reqs[1].UpdateSheetProperties.Properties.Title)
	require.Equal(t, wantNewID, reqs[1].UpdateSheetProperties.Properties.SheetID, "改的是新表不是模板")

	// ③ 改表内第 1 行标题：值逐字相等，且写的确实是 A1 那一格
	require.NotNil(t, reqs[2].UpdateCells, "第 3 步必须是 updateCells")
	require.Equal(t, wantNewID, reqs[2].UpdateCells.Range.SheetID)
	require.Equal(t, int64(0), reqs[2].UpdateCells.Range.StartRowIndex)
	require.Equal(t, int64(1), reqs[2].UpdateCells.Range.EndRowIndex)
	require.Equal(t, "userEnteredValue", reqs[2].UpdateCells.Fields)
	require.Len(t, reqs[2].UpdateCells.Rows, 1)
	require.Len(t, reqs[2].UpdateCells.Rows[0].Values, 1)
	require.Equal(t, "2021 年 · 金融数据追踪",
		reqs[2].UpdateCells.Rows[0].Values[0].UserEnteredValue.StringValue,
		"只换年份、保留后半句（逐字相等，不是 Contains）")

	// ④ 排位
	require.NotNil(t, reqs[3].UpdateSheetProperties, "第 4 步必须是改属性")
	require.Equal(t, "index", reqs[3].UpdateSheetProperties.Fields)
	require.Equal(t, int64(3), reqs[3].UpdateSheetProperties.Properties.Index)
	require.Equal(t, wantNewID, reqs[3].UpdateSheetProperties.Properties.SheetID)
}

// W5 —— index 边界：最前（新年份比谁都小）与最末（比谁都大）。
// 注意非年度表「说明」占着表序 0，所以「排最前」的正确答案是 1 而不是 0。
func TestW_PushPlacesNewTabAtEdges(t *testing.T) {
	t.Run("比现有都小则排在第一张年度表之前", func(t *testing.T) {
		c, rec := newTestClient(t,
			vResp("2024 年 · 金融数据追踪", []string{"说明", "2024年", "2026年"}, "2019年"), 0)
		_, err := Push(context.Background(), c, rowsSpanning(2019, 2019), sampleLabels(),
			Options{Apply: true, CreateSheets: true})
		require.NoError(t, err)
		require.Equal(t, []int64{1}, vNewTabIndexes(t, rec), "第一个 > 2019 的是 2024年，在表序第 1 位")
	})
	t.Run("比现有都大则排末尾", func(t *testing.T) {
		c, rec := newTestClient(t,
			vResp("2024 年 · 金融数据追踪", []string{"说明", "2024年", "2026年"}, "2030年"), 0)
		_, err := Push(context.Background(), c, rowsSpanning(2030, 2030), sampleLabels(),
			Options{Apply: true, CreateSheets: true})
		require.NoError(t, err)
		require.Equal(t, []int64{3}, vNewTabIndexes(t, rec), "没有 > 2030 的，排末尾 = len(order) = 3")
	})
}

// W6 —— 连建两张：后一张的 index 必须把前一张已插入的位置算进去。
// 现有 {说明,2024年,2026年} 缺 {2023年,2025年}：
//
//	2023 ⇒ 第一个 >2023 的是 2024年（位置 1）⇒ index 1，插入后表序 [说明,2023年,2024年,2026年]
//	2025 ⇒ 第一个 >2025 的是 2026年（**现在**在位置 3）⇒ index 3
//
// 若实现忘了把前一张插进本地表序，第二张会算成 2 —— 这正是本条要钉住的。
func TestW_PushPlacesTwoNewTabsCumulatively(t *testing.T) {
	c, rec := newTestClient(t,
		vResp("2024 年 · 金融数据追踪", []string{"说明", "2024年", "2026年"}, "2023年", "2025年"), 0)

	_, err := Push(context.Background(), c, rowsSpanning(2023, 2026), sampleLabels(),
		Options{Apply: true, CreateSheets: true})
	require.NoError(t, err)
	require.Equal(t, []int64{1, 3}, vNewTabIndexes(t, rec),
		"第二张的 index 必须把第一张已插入的位置算进去")
}

// W7 —— 建表必须先于写数据，且新建表的行进了写请求。
func TestW_PushCreatesBeforeWritingAndWritesNewTab(t *testing.T) {
	c, rec := newTestClient(t,
		vResp("2024 年 · 金融数据追踪", []string{"说明", "2024年", "2026年"}, "2023年", "2025年"), 0)
	rows := rowsSpanning(2023, 2026)

	res, err := Push(context.Background(), c, rows, sampleLabels(), Options{Apply: true, CreateSheets: true})
	require.NoError(t, err)
	require.Equal(t, []string{"2023年", "2025年"}, res.MissingTabs)

	// 顺序：两个建表 batchUpdate 都要早于唯一的数据写请求。
	var iDup []int
	iWrite := -1
	for i, m := range rec.methods {
		if strings.HasPrefix(m, "GET ") {
			continue
		}
		if m == "POST "+pathBatchUpdate {
			iDup = append(iDup, i)
		} else if strings.Contains(rec.bodies[i], "valueInputOption") {
			iWrite = i
		}
	}
	require.Len(t, iDup, 2, "缺两张建两张")
	require.NotEqual(t, -1, iWrite, "必须有一个写数据请求")
	for _, i := range iDup {
		require.Less(t, i, iWrite, "建表必须先于写数据，否则 range 指向不存在的表")
	}

	// 新表录入区全空 ⇒ 它们的行全是 WillWrite；写体里必须出现新表的 range。
	requireInvariant(t, res, rows, sampleLabels())
	var got batchBody
	require.NoError(t, json.Unmarshal([]byte(lastWriteBody(t, rec)), &got))
	require.Len(t, got.Data, res.WillWrite, "写的格数必须等于 WillWrite")
	var ranges []string
	for _, d := range got.Data {
		ranges = append(ranges, d.Range)
	}
	require.Contains(t, ranges, "'2023年'!A15", "新建表的行也要写进去")
	require.Contains(t, ranges, "'2025年'!A15")
}
